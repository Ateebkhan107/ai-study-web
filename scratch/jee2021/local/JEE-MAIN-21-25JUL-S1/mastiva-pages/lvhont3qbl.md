1.   $(2 - \sqrt{3}) - \log_e 2$
2.   $(2 + \sqrt{3}) + \log_e 2$
3.   $(1 + \sqrt{3}) - \log_e(\sqrt{3} - 1)$
4.   $(1 - \sqrt{3}) - \log_e(\sqrt{3} - 1)$
